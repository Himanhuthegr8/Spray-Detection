void mosaic_display_save(const int _x_m, const int _y_m, Mat* _img_m, int show_flag, int save_flag, int _id) {
    //namedWindow("Cluster_Result", 0);

    const int single_frame_width = _img_m[0].cols, single_frame_height = _img_m[0].rows;
    const int Mosaic_width = single_frame_width * _x_m, Mosaic_height = single_frame_height * _y_m;
    Mat All_Results(Mosaic_height, Mosaic_width, CV_8UC3);
    char szSave[10][200];
    int k = 0;

    for (int i = 0; i < _x_m * _y_m; i++) {
        rectangle(_img_m[i], Point(0, 0), Point(_img_m->cols, _img_m->rows), Scalar(255, 255, 255), 1, 1);
    }

    for (int i = 0; i < _x_m * _y_m; i++) {
        if (_img_m[i].channels() == 1) cvtColor(_img_m[i], _img_m[i], COLOR_GRAY2BGR);
    }
    for (int h_pos = 0; h_pos < _y_m; h_pos++) {
        for (int w_pos = 0; w_pos < _x_m; w_pos++) {
            _img_m[k].copyTo(All_Results
                .rowRange(h_pos * single_frame_height, h_pos * single_frame_height + single_frame_height)
                .colRange(w_pos * single_frame_width, w_pos * single_frame_width + single_frame_width));

            k++;
        }
    }

    if (save_flag == 1) {
        sprintf_s(szSave[0], sizeof(szSave[0]), "All_Images/seg_%04d.png", _id); //"All_Images/All_Img_%04d.png
        cv::imwrite(szSave[0], All_Results);
    }

    if (show_flag == 1) cv::imshow("Result", All_Results);
}

std::vector<int>get_largest_contour(Mat in_img, Mat in_gray_img, Mat out_color_img, Mat out_color_filled_img, Mat out_color_cropped_img) {
    std::vector<int> array_data(4);
    vector<vector<Point>> contours;
    std::vector<cv::Vec4i> hierarchy;
    int largest_area = 0, largest_contour_index = 0;

    cv::findContours(in_gray_img.clone(), contours, hierarchy, cv::RETR_TREE, cv::CHAIN_APPROX_NONE);

    for (int i = 0; i < contours.size(); ++i)
    {
        double a = contourArea(contours[i], false);  //Find the area of contour
        if (a > largest_area) largest_area = a, largest_contour_index = i;
    }
    vector<vector<Point> > contours_poly(contours.size());
    vector<Rect> boundRect(contours.size());
    if (contours.size() != 0) {
        cv::drawContours(out_color_img, contours, largest_contour_index, Scalar(255, 255, 0), 1, 8, hierarchy, 0, cv::Point()); // get max area boundary
        cv::drawContours(out_color_filled_img, contours, largest_contour_index, Scalar(255, 255, 255), -1, 8, hierarchy, 0, cv::Point()); // get max area boundary
        approxPolyDP(Mat(contours[largest_contour_index]), contours_poly[largest_contour_index], 3, true);
        boundRect[largest_contour_index] = boundingRect(Mat(contours_poly[largest_contour_index]));
        //cv::rectangle(out_color_cropped_img, boundRect[largest_contour_index].tl(), boundRect[largest_contour_index].br(), Scalar(0, 0, 255), 2, 8, 0); //Draws the actual rectangle around the contours

        array_data[0] = (int)boundRect[largest_contour_index].tl().x;
        array_data[1] = (int)boundRect[largest_contour_index].tl().y;
        array_data[2] = (int)boundRect[largest_contour_index].br().x;
        array_data[3] = (int)boundRect[largest_contour_index].br().y;
    }


    Mat contour_gray_img;
    cv::cvtColor(out_color_filled_img, contour_gray_img, COLOR_BGR2GRAY);
    for (int y = 0; y < contour_gray_img.rows; y++) {
        for (int x = 0; x < contour_gray_img.cols; x++) {
            if (contour_gray_img.data[y * contour_gray_img.cols + x] == 255) {
                out_color_cropped_img.at<Vec3b>(Point(x, y)) = in_img.at<Vec3b>(Point(x, y));
            }
        }
    }
    //----------crop input image using contour

    return array_data;
}

void  get_flowAnalysis_cropped_boundary(int _id, Mat curr_img, Mat pre_img, Mat contour_img, Mat of_img, Mat avg_of_img, int cell, int _arrow, int type) {
    const int buff = 7;
    UINT32 avg_vector[4][buff];

    Mat bn_flow(curr_img.size(), CV_32FC2);
    Point bn_ptBegin, bn_ptEnd, bn_ptVec, bn_ptAllowCW, bn_ptAllowCCW;
    Point ptEnd, ptVec;

    cvtColor(curr_img, curr_img, COLOR_BGR2GRAY);
    cvtColor(pre_img, pre_img, COLOR_BGR2GRAY);


    calcOpticalFlowFarneback(pre_img, curr_img, bn_flow, 0.1 * py_scale, levels, W_Size, iterations, 5, 1.2, 0);

    for (int b = 0; b < buff; b++) {
        avg_vector[0][b] = 0; avg_vector[1][b] = 0; avg_vector[2][b] = 0; avg_vector[3][b] = 0;
    }

    Mat contour_gray_img;
    cv::cvtColor(contour_img, contour_gray_img, COLOR_BGR2GRAY);
    for (int y = 0; y < contour_gray_img.rows; y += cell) {
        for (int x = 0; x < contour_gray_img.cols; x += cell) {
            if (contour_gray_img.data[y * contour_gray_img.cols + x] == 255) {
                //  const Point2f bn_flowatxy, flowatxy;
                if (type == 0) _arrow = 5;
                else _arrow = 5;
                Point2f bn_flowatxy = bn_flow.at<Point2f>(y, x) * _arrow;
                const Point2f flowatxy = bn_flow.at<Point2f>(y, x);
                bn_ptBegin.x = x;
                bn_ptBegin.y = y;

                ptEnd.x = cvRound(x + flowatxy.x);
                ptEnd.y = cvRound(y + flowatxy.y);

                bn_ptEnd.x = cvRound(x + bn_flowatxy.x);
                bn_ptEnd.y = cvRound(y + bn_flowatxy.y);

                bn_ptVec.x = bn_ptEnd.x - bn_ptBegin.x;
                bn_ptVec.y = bn_ptEnd.y - bn_ptBegin.y;

                bn_ptVec.x = bn_ptVec.x / 2;
                bn_ptVec.y = bn_ptVec.y / 2;
                if (bn_ptVec.x != 0 || bn_ptVec.y != 0) {
                    double bn_SinTheta = sqrt(2.) / 2.;
                    double bn_CosTheta = sqrt(2.) / 2.;
                    bn_ptAllowCW.x = (int)(-1 / 2. * bn_ptVec.x * bn_CosTheta + -1 / 2. * bn_ptVec.y * bn_SinTheta + 0.5);
                    bn_ptAllowCW.y = (int)(-1 / 2. * bn_ptVec.x * -bn_SinTheta + -1 / 2. * bn_ptVec.y * bn_CosTheta + 0.5);
                    bn_ptAllowCCW.x = (int)(-1 / 2. * bn_ptVec.x * bn_CosTheta - -1 / 2. * bn_ptVec.y * bn_SinTheta + 0.5);
                    bn_ptAllowCCW.y = (int)(-1 / 2. * bn_ptVec.x * bn_SinTheta + -1 / 2. * bn_ptVec.y * bn_CosTheta + 0.5);

                    //-------if scene is static
                    circle(of_img, bn_ptBegin, 1, Scalar(0, 255, 0), 2);
                    cv::line(of_img, bn_ptEnd, bn_ptBegin, CV_RGB(255, 0, 0), 2);
                    cv::line(of_img, bn_ptEnd, Point(bn_ptEnd.x + bn_ptAllowCW.x, bn_ptEnd.y + bn_ptAllowCW.y), CV_RGB(255, 0, 0), 2);
                    cv::line(of_img, bn_ptEnd, Point(bn_ptEnd.x + bn_ptAllowCCW.x, bn_ptEnd.y + bn_ptAllowCCW.y), CV_RGB(255, 0, 0), 2);
                    //-------if scene is static
                }

            }
        }
    }

}

void alignImages(Mat& im1, Mat& im2, Mat& im1Reg, Mat& h)
{
    // Convert images to grayscale
    Mat im1Gray, im2Gray;
    cvtColor(im1, im1Gray, cv::COLOR_BGR2GRAY);
    cvtColor(im2, im2Gray, cv::COLOR_BGR2GRAY);

    // Variables to store keypoints and descriptors
    std::vector<KeyPoint> keypoints1, keypoints2;
    Mat descriptors1, descriptors2;

    // Detect ORB features and compute descriptors.
    Ptr<Feature2D> orb = ORB::create(MAX_FEATURES);
    orb->detectAndCompute(im1Gray, Mat(), keypoints1, descriptors1);
    orb->detectAndCompute(im2Gray, Mat(), keypoints2, descriptors2);

    // Match features.
    std::vector<DMatch> matches;
    Ptr<DescriptorMatcher> matcher = DescriptorMatcher::create("BruteForce-Hamming");
    matcher->match(descriptors1, descriptors2, matches, Mat());

    // Sort matches by score
    std::sort(matches.begin(), matches.end());

    // Remove not so good matches
    const int numGoodMatches = matches.size() * GOOD_MATCH_PERCENT;
    matches.erase(matches.begin() + numGoodMatches, matches.end());

    // Draw top matches
    Mat imMatches;
    drawMatches(im1, keypoints1, im2, keypoints2, matches, imMatches);
    //imwrite("matches.jpg", imMatches);
    imshow("matched", imMatches);

    // Extract location of good matches
    std::vector<Point2f> points1, points2;

    for (size_t i = 0; i < matches.size(); i++)
    {
        points1.push_back(keypoints1[matches[i].queryIdx].pt);
        points2.push_back(keypoints2[matches[i].trainIdx].pt);
    }

    // Find homography
    h = findHomography(points1, points2, RANSAC);

    // Use homography to warp image
    warpPerspective(im1, im1Reg, h, im2.size());
}




void ContourMatch(Mat pre_img, Mat cur_img)
{

    cv::Mat dst = pre_img.clone();


    if (pre_img.channels() == 3) cvtColor(pre_img, pre_img, COLOR_BGR2GRAY);
    if (cur_img.channels() == 3) cvtColor(cur_img, cur_img, COLOR_BGR2GRAY);


    cv::Mat temp_bin, src_bin;
    cv::threshold(pre_img, temp_bin, tempThres, 255, cv::THRESH_BINARY_INV);
    cv::threshold(cur_img, src_bin, srcThres, 255, cv::THRESH_BINARY_INV);

    cv::morphologyEx(temp_bin, temp_bin, cv::MORPH_OPEN, cv::Mat(), cv::Point(-1, -1), 2);
    cv::morphologyEx(src_bin, src_bin, cv::MORPH_OPEN, cv::Mat(), cv::Point(-1, -1), 2);

    cv::imshow("temp_bin", temp_bin);
    cv::Mat labelsImg;
    cv::Mat stats;
    cv::Mat centroids;
    int nLabels = cv::connectedComponentsWithStats(src_bin, labelsImg, stats, centroids);

    cv::Mat roiImg;
    cv::cvtColor(src_bin, roiImg, COLOR_GRAY2BGR);
    std::vector<cv::Rect> roiRects;
    for (int i = 1; i < nLabels; i++) {
        int* param = stats.ptr<int>(i);

        int x = param[cv::ConnectedComponentsTypes::CC_STAT_LEFT];
        int y = param[cv::ConnectedComponentsTypes::CC_STAT_TOP];
        int height = param[cv::ConnectedComponentsTypes::CC_STAT_HEIGHT];
        int width = param[cv::ConnectedComponentsTypes::CC_STAT_WIDTH];
        roiRects.push_back(cv::Rect(x, y, width, height));

        cv::rectangle(roiImg, roiRects.at(i - 1), cv::Scalar(0, 255, 0), 2);
    }
    cv::imshow("roiImg", roiImg);

    cout << "nLabels=" << nLabels << endl;

    // huモーメントによる形状マッチングを行う
    for (int i = 1; i < nLabels; i++) {
        cv::Mat roi = src_bin(roiRects.at(i - 1));    // 対象とするブロブの領域取り出し
        double similarity = cv::matchShapes(temp_bin, roi, CONTOURS_MATCH_I1, 0);    // huモーメントによるマッチング

       // cout << "similarity="<< similarity << endl;

        if (similarity < huMomentThres) {
            cv::rectangle(dst, roiRects.at(i - 1), cv::Scalar(0, 255, 0), 2);
        }
    }

    //  cv::imshow("template", temp);
    cv::imshow("dst", dst);
    // cv::imshow("dst", cur_img);
    // cv::waitKey(1);

     //cv::imwrite("dst.jpg", dst);

}
